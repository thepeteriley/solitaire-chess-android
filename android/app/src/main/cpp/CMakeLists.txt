cmake_minimum_required(VERSION 3.22.1)
project(solitaire_stockfish LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(SF "${CMAKE_CURRENT_LIST_DIR}/../../../../Vendor/Stockfish/src")
file(GLOB_RECURSE SF_SOURCES CONFIGURE_DEPENDS "${SF}/*.cpp")
list(FILTER SF_SOURCES EXCLUDE REGEX "/main\\.cpp$|/universal/")
add_library(solitaire_stockfish SHARED stockfish_jni.cpp ${SF_SOURCES})
target_include_directories(solitaire_stockfish PRIVATE "${SF}")
target_compile_definitions(solitaire_stockfish PRIVATE NNUE_EMBEDDING_OFF=1 SOLITAIRE_NO_SHM=1 IS_64BIT=1)
if(ANDROID_ABI STREQUAL "arm64-v8a")
  target_compile_definitions(solitaire_stockfish PRIVATE USE_NEON=1)
endif()
target_compile_options(solitaire_stockfish PRIVATE -O3 -fexceptions -frtti)
target_link_options(solitaire_stockfish PRIVATE "-Wl,-z,max-page-size=16384")
